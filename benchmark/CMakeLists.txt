# This function controls experiment parameters, not optimization or build mode.
function(fh_index_add_benchmark SOURCE LATENCY VERSION BLOCK_SIZE FINGER_SPARSITY)
    get_filename_component(STEM "${SOURCE}" NAME_WE)
    set(NAME "${STEM}_lat${LATENCY}_type${VERSION}_b${BLOCK_SIZE}_f${FINGER_SPARSITY}")
    fh_index_add_program(${NAME} "${SOURCE}")
    target_compile_definitions(${NAME} PRIVATE
        BLOCK_SIZE=${BLOCK_SIZE} FINGER_SPARSITY=${FINGER_SPARSITY})
    if(LATENCY)
        target_compile_definitions(${NAME} PRIVATE LATENCY=1)
    endif()
    if(VERSION STREQUAL "RO" OR VERSION STREQUAL "RW")
        target_compile_definitions(${NAME} PRIVATE VERSION_${VERSION})
    elseif(NOT VERSION STREQUAL "NONE")
        message(FATAL_ERROR "Benchmark version must be RO, RW or NONE")
    endif()
    if(FH_INDEX_RECLAMATION_DEFINITION)
        target_compile_definitions(${NAME} PRIVATE ${FH_INDEX_RECLAMATION_DEFINITION})
    endif()
    if(FH_INDEX_ALLOCATOR STREQUAL "jemalloc")
        target_link_libraries(${NAME} PRIVATE "${FH_INDEX_JEMALLOC_LIBRARY}")
        target_compile_definitions(${NAME} PRIVATE FH_INDEX_USE_JEMALLOC=1)
    endif()
endfunction()

set(LATENCY_VALUES 0 CACHE STRING "Benchmark latency modes")
set(BLOCK_SIZES "32;40;48" CACHE STRING "Benchmark block sizes")
set(FINGER_SPARSITIES "2;4" CACHE STRING "Benchmark finger sparsities")
set(VERSION_TYPES "RO;RW" CACHE STRING "Benchmark index versions")

foreach(LATENCY IN LISTS LATENCY_VALUES)
    foreach(VERSION IN LISTS VERSION_TYPES)
        foreach(BLOCK_SIZE IN LISTS BLOCK_SIZES)
            foreach(FINGER_SPARSITY IN LISTS FINGER_SPARSITIES)
                fh_index_add_benchmark(bench_fh_index.cpp
                    ${LATENCY} ${VERSION} ${BLOCK_SIZE} ${FINGER_SPARSITY})
            endforeach()
        endforeach()
    endforeach()
endforeach()

foreach(BENCHMARK IN ITEMS bench_segmenting bench_segmenting_effect
        bench_alignment_stats bench_malloc bench_epoch_reclamation)
    fh_index_add_benchmark(${BENCHMARK}.cpp 0 NONE 0 0)
endforeach()
fh_index_add_benchmark(bench_insert_breakdown.cpp 0 NONE 40 2)
